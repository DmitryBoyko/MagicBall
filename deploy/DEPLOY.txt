## ПЕРВЫЙ ДЕПЛОЙ
══════════════════════════════════════════════════════════════

### 1. ПК
  cd C:\epm-games\MagicalBall
  .\deploy\prepare-winscp.ps1

### 2. WinSCP
  deploy\winscp-upload\*  →  /opt/magicalball/
  (должен быть /opt/magicalball/docker-compose.yml)
  НЕ перезаписывать .env при следующих заливках.

### 3. VPS — ОДНА КОМАНДА
  mkdir -p /opt/magicalball
  cd /opt/magicalball
  chmod +x deploy/*.sh
  bash deploy/setup-vps.sh

  setup-vps.sh сам:
    - создаёт .env
    - VPS_IP=147.45.173.26, API_PORT=18437
    - копирует GIGACHAT_CREDENTIALS из /opt/truetaro/.env
    - ufw allow 18437/tcp (если ufw активен)
    - docker compose up + health

### 3-alt. Ручной блок (если нет setup-vps.sh)
  cd /opt/magicalball
  cp -n deploy/env.example .env
  sed -i 's/^VPS_IP=.*/VPS_IP=147.45.173.26/' .env
  sed -i 's/^API_PORT=.*/API_PORT=18437/' .env
  CRED=$(grep '^GIGACHAT_CREDENTIALS=' /opt/truetaro/.env | cut -d= -f2- | tr -d '\r')
  sed -i "s|^GIGACHAT_CREDENTIALS=.*|GIGACHAT_CREDENTIALS=${CRED}|" .env
  chmod +x deploy/*.sh
  ufw allow 18437/tcp 2>/dev/null || true
  bash deploy/install-on-vps.sh

### 4. Проверка
  curl -s http://127.0.0.1:18437/health
  curl -s http://147.45.173.26:18437/health
  # {"status":"ok","service":"magicalball"}

  ss -tlnp | grep ':18437'
  docker compose ps

### 5. Firewall (оба места!)
  ufw allow 18437/tcp          # на VPS
  панель хостера → TCP 18437   # иначе с телефона не достучаться

### 6. Godot / APK
  proxy_base_url:    http://127.0.0.1:18437      (редактор)
  android_base_url:  http://147.45.173.26:18437  (APK)
  После смены api.json → пересобрать APK
  С телефона: одно гадание → ответ LLM, не туман
